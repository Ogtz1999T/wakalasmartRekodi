function switchMainTab(tab) { // Kama ni 'admin' na mtumiaji sio admin, mzuie kabisa if (tab === 'admin' && currentUser.role !== 'admin') { alert('Huruhusiwi kuingia hapa!'); return; } const btnNew = document.getElementById('mainTabNew'); const btnHistory = document.getElementById('mainTabHistory'); const btnAdmin = document.getElementById('mainTabAdmin'); const viewNew = document.getElementById('viewNew'); const viewHistory = document.getElementById('viewHistory'); const viewAdmin = document.getElementById('viewAdmin'); btnNew.className = "px-4 py-2 text-sm font-semibold text-slate-500 hover:text-blue-900 flex items-center gap-1.5"; btnHistory.className = "px-4 py-2 text-sm font-semibold text-slate-500 hover:text-blue-900 flex items-center gap-1.5"; if(btnAdmin && currentUser.role === 'admin') { btnAdmin.className = "px-4 py-2 text-sm font-semibold text-amber-800 flex items-center gap-1.5"; } viewNew.classList.add('hidden'); viewHistory.classList.add('hidden'); if(viewAdmin) viewAdmin.classList.add('hidden'); if (tab === 'new') { btnNew.className = "px-4 py-2 text-sm font-semibold text-blue-950 border-b-2 border-blue-950 flex items-center gap-1.5"; viewNew.classList.remove('hidden'); } else if (tab === 'history') { btnHistory.className = "px-4 py-2 text-sm font-semibold text-blue-950 border-b-2 border-blue-950 flex items-center gap-1.5"; viewHistory.classList.remove('hidden'); loadHistory(); } else if (tab === 'admin' && currentUser.role === 'admin') { btnAdmin.className = "px-4 py-2 text-sm font-semibold text-amber-600 border-b-2 border-amber-600 flex items-center gap-1.5"; viewAdmin.classList.remove('hidden'); loadAdminUsers(); } }
